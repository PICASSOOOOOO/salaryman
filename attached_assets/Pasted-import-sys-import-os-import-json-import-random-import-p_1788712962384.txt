# --- 2. NOTIFICATION SUB-ENGINE ---
class ToastNotification:
    def __init__(self, message, style="warning"):
        self.message = message
        self.style = style
        self.lifetime = 3.5
        self.alpha = 255

    def update(self, dt):
        self.lifetime -= dt
        if self.lifetime < 1.0:
            self.alpha = max(0, int(self.lifetime * 255))
        return self.lifetime > 0

# --- 3. HIGH-PERFORMANCE CENTRAL GAME STATE MANAGER ---
class OfficeManager:
    def __init__(self):
        self.money = 1500.0
        self.current_hour = 9
        self.current_day = 1
        self.tick_timer = 0.0
        self.seconds_per_hour = 2.0
        self.employees = []
        self.alerts = []
        self.shop_candidates = []

    def push_alert(self, text, style="warning"):
        if not any(a.message == text for a in self.alerts):
            self.alerts.insert(0, ToastNotification(text, style))

    def update(self, delta_time):
        self.tick_timer += delta_time
        if self.tick_timer >= self.seconds_per_hour:
            self.tick_timer = 0.0
            self.advance_time()
        self.alerts = [a for a in self.alerts if a.update(delta_time)]

    def advance_time(self):
        self.current_hour += 1
        is_shift_active = 9 <= self.current_hour <= 17

        for emp in self.employees:
            emp.hourly_update(self.current_hour, is_shift_active)

        if self.current_hour > 17:
            self.current_hour = 9
            self.current_day += 1
            self.save_game_state()  # Auto-save at the end of every business day
            self.push_alert(f"Day {self.current_day - 1} complete. Auto-saved data.", "success")

    def refresh_shop_candidates(self):
        self.shop_candidates = random.sample(CANDIDATE_POOL, 3)

    # --- ENHANCED STORAGE ENGINE ---
    def save_game_state(self):
        save_data = {
            "money": self.money,
            "current_day": self.current_day,
            "current_hour": self.current_hour,
            "employees": [
                {"name": e.name, "prod": e.base_productivity, "wage": e.salary_per_day, "stamina": e.stamina}
                for e in self.employees
            ]
        }
        try:
            with open(SAVE_FILE, "w") as f:
                json.dump(save_data, f)
        except Exception as e:
            print(f"Save Failure: {e}")

    def load_game_state(self):
        if not os.path.exists(SAVE_FILE):
            # Seed default starter staff if no save module file exists
            self.employees = [EmployeeCard("John Doe", 15.0, 70.0), EmployeeCard("Jane Smith", 28.0, 130.0)]
            self.refresh_shop_candidates()
            return
        try:
            with open(SAVE_FILE, "r") as f:
                data = json.load(f)
                self.money = data.get("money", 1500.0)
                self.current_day = data.get("current_day", 1)
                self.current_hour = data.get("current_hour", 9)
                self.employees = []
                for emp_data in data.get("employees", []):
                    emp = EmployeeCard(emp_data["name"], emp_data["prod"], emp_data["wage"])
                    emp.stamina = emp_data["stamina"]
                    self.employees.append(emp)
            self.push_alert("Loaded previous enterprise save file.", "success")
        except Exception as e:
            self.push_alert("Failed to load secure save data.", "danger")
            self.employees = [EmployeeCard("John Doe", 15.0, 70.0)]
        self.refresh_shop_candidates()

    def draw_alerts(self, surface):
        start_y = HEIGHT - 60
        for i, alert in enumerate(self.alerts[:4]):
            alert_rect = pygame.Rect(WIDTH - 330, start_y - (i * 55), 300, 45)
            s = pygame.Surface((alert_rect.width, alert_rect.height), pygame.SRCALPHA)
            
            if alert.style == "danger":       col = (239, 68, 68, alert.alpha)
            elif alert.style == "success":   col = (16, 185, 129, alert.alpha)
            else:                            col = (245, 158, 11, alert.alpha)
            
            pygame.draw.rect(s, col, (0, 0, alert_rect.width, alert_rect.height), border_radius=6)
            surface.blit(s, (alert_rect.x, alert_rect.y))
            lbl_msg = FONT_UI_REG.render(alert.message, True, (255, 255, 255))
            lbl_msg.set_alpha(alert.alpha)
            surface.blit(lbl_msg, (alert_rect.x + 15, alert_rect.y + 13))

class EmployeeCard:
    def __init__(self, name, productivity, salary):
        self.name = name
        self.base_productivity = productivity
        self.salary_per_day = salary
        self.width, self.height = 280, 150
        self.coffee_btn = pygame.Rect(0, 0, 85, 28)
        self.max_stamina = 100.0
        self.stamina = 100.0
        self.drain_rate = 7.0
        self.coffee_cost = 25.0
        self.is_resting = False

    def hourly_update(self, hour, is_shift_active):
        if is_shift_active:
            if self.is_resting:
                self.stamina = min(self.max_stamina, self.stamina + 40.0)
                if self.stamina >= self.max_stamina:
                    self.is_resting = False
            else:
                if self.stamina > 0:
                    self.stamina = max(0.0, self.stamina - self.drain_rate)
                    efficiency = self.stamina / self.max_stamina
                    manager.money += (self.base_productivity * efficiency)
                    
                    if self.stamina <= 0.0:
                        manager.push_alert(f"{self.name} is EXHAUSTED!", "danger")
                    elif self.stamina <= 20.0:
                        manager.push_alert(f"{self.name} needs a coffee break.", "warning")
        if hour == 17:
            manager.money -= self.salary_per_day
