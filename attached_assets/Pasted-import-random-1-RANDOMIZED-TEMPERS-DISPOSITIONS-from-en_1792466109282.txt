import random

# 🧠 1. RANDOMIZED TEMPERS & DISPOSITIONS (from enRage: Hothead, Coward, Loyal)
PERSONALITY_TRAITS = {
    "Gossip": {"grudge_multiplier": 1.5, "stamina_drain": 1.0, "desc": "Spreads rumors faster; magnifies corporate grudges."},
    "Sycophant": {"grudge_multiplier": 0.8, "stamina_drain": 1.2, "desc": "Protects the manager; harder to make a true nemesis."},
    "Workaholic": {"grudge_multiplier": 1.0, "stamina_drain": 0.5, "desc": "High physical resilience; loses stamina much slower."},
    "Saboteur": {"grudge_multiplier": 1.3, "stamina_drain": 1.0, "desc": "Naturally aggressive; escalates to active sabotage instantly."}
}

class DepartmentFaction:
    """🏢 CORPORATE FACTIONS (Replicating enRage's 8 Outlaw Outfits/Gangs)"""
    def __init__(self, name: str):
        self.name = name
        self.department_morale = 100  # Drops if you attack members of this department
        self.budget_treasury = 5000   # Corporate assets that can be sabotaged

class CoworkerNPC:
    def __init__(self, name: str, department: DepartmentFaction):
        self.name = name
        self.department = department  # Binds NPC to a broader faction gang
        self.stamina = 100
        self.grudge_score = 0         # Escales: Resentful -> Vengeful -> Murderous (Corporate equivalent)
        self.corporate_rank = "Associate"  # Hierarchy rank: Associate -> Manager -> Director
        
        # Pull random life story/temper parameters just like enRage
        self.trait_name = random.choice(list(PERSONALITY_TRAITS.keys()))
        self.trait_config = PERSONALITY_TRAITS[self.trait_name]
        
        # Memory Vault explicitly logs details (e.g., event type, physical cost, department impact)
        self.memory_vault = [] 
        self.state = "Idle"

    def log_interaction(self, event_type: str, raw_impact: int):
        """Logs the encounter, applies trait multipliers, and hits department faction morale."""
        # Multiplier from their random personality quirk
        final_impact = int(raw_impact * self.trait_config["grudge_multiplier"])
        self.grudge_score += final_impact
        
        # Action echoes to the entire department faction (Inter-department warfare)
        if final_impact > 15:
            self.department.department_morale -= int(final_impact * 0.5)
            
        self.memory_vault.append({"action": event_type, "severity": final_impact})
        self.update_state()

    def update_state(self):
        """Finite State Machine matching enRage's escalation parameters."""
        if self.stamina <= 0:
            self.state = "Burnout"
        elif self.grudge_score >= 60 or self.department.department_morale < 40:
            self.state = "Corporate_Warfare"  # Equal to enRage's 'Murderous' state
        elif self.grudge_score >= 25:
            self.state = "Passive_Aggressive" # Equal to enRage's 'Resentful' state
        else:
            self.state = "Idle"

    def execute_office_action(self) -> str:
        """Emergent narrative generation using traits and states."""
        status_prefix = f"[{self.name} | {self.department.name} {self.corporate_rank} | Quirks: {self.trait_name}]"
        
        if self.state == "Corporate_Warfare":
            return f"⚠️ {status_prefix} has declared war! They filed a formal compliance complaint and drained ${random.randint(500, 1000)} from your project budget!"
        elif self.state == "Passive_Aggressive":
            return f"💬 {status_prefix} left a sticky note on your monitor: 'Please see me regarding your workflow transparency.'"
        elif self.state == "Burnout":
            return f"💤 {status_prefix} is completely burned out from your demands. Out of office auto-responder active."
        return f"💻 {status_prefix} is processing standard spreadsheets."
