# --- 2. FLOATING ALERTS ENGINE ---
class ToastNotification:
    def __init__(self, message, type="warning"):
        self.message = message
        self.type = type
        self.lifetime = 3.5  # Seconds toast stays active
        self.alpha = 255     # Fade opacity mapping

    def update(self, dt):
        self.lifetime -= dt
        if self.lifetime < 1.0:
            self.alpha = max(0, int(self.lifetime * 255))
        return self.lifetime > 0


# --- 3. CORE SYSTEM LOGIC MANAGER ---
class OfficeManager:
    def __init__(self):
        self.money = 2500.0
        self.current_hour = 9
        self.current_day = 1
        self.tick_timer = 0.0
        self.seconds_per_hour = 2.0
        self.employees = []
        self.alerts = []  # Active notification track

    def push_alert(self, text, type="warning"):
        # Enforce unique alerts to avoid spamming the log
        if not any(a.message == text for a in self.alerts):
            self.alerts.insert(0, ToastNotification(text, type))

    def update(self, delta_time):
        self.tick_timer += delta_time
        if self.tick_timer >= self.seconds_per_hour:
            self.tick_timer = 0.0
            self.advance_time()
            
        # Update running notification animations
        self.alerts = [a for a in self.alerts if a.update(delta_time)]

    def advance_time(self):
        self.current_hour += 1
        is_shift_active = 9 <= self.current_hour <= 17

        for emp in self.employees:
            emp.hourly_update(self.current_hour, is_shift_active)

        if self.current_hour > 17:
            self.current_hour = 9
            self.current_day += 1

    def draw_alerts(self, surface):
        # Render stack starting from the lower right canvas quadrant
        start_y = HEIGHT - 50
        for i, alert in enumerate(self.alerts[:4]):  # Cap visible queue size to 4
            alert_rect = pygame.Rect(WIDTH - 320, start_y - (i * 55), 300, 45)
            
            # Alpha surfaces for smooth fade-out animations
            s = pygame.Surface((alert_rect.width, alert_rect.height), pygame.SRCALPHA)
            bg_col = (239, 68, 68, alert.alpha) if alert.type == "danger" else (245, 158, 11, alert.alpha)
            
            pygame.draw.rect(s, bg_col, (0, 0, alert_rect.width, alert_rect.height), border_radius=6)
            surface.blit(s, (alert_rect.x, alert_rect.y))
            
            txt_color = (255, 255, 255)
            lbl_msg = FONT_UI_REG.render(alert.message, True, txt_color)
            lbl_msg.set_alpha(alert.alpha)
            surface.blit(lbl_msg, (alert_rect.x + 15, alert_rect.y + 13))

# --- 5. EXECUTION & APP PIPELINE ---
manager = OfficeManager()
scroll_y = 0
max_scroll = 0

# Bulk Mock Data population verifying Grid Scalability properties
worker_roster = [
    ("Alice Vance", 30, 120), ("Bob Wright", 45, 180), ("Charlie Fox", 60, 250),
    ("David Miller", 25, 100), ("Eva Green", 50, 210), ("Frank Castle", 80, 400),
    ("Grace Hopper", 95, 500), ("Henry Ford", 40, 160), ("Ivy League", 70, 310)
]
for item in worker_roster:
    manager.employees.append(EmployeeCard(item[0], item[1], item[2]))

running = True
while running:
    dt = clock.tick(60) / 1000.0
    
    # Process Inputs
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
            
        # Vertical Scrolling Calculations mapping mouse wheel mechanics
        elif event.type == pygame.MOUSEBUTTONDOWN:
            if event.button == 4:    # Scroll Up
                scroll_y = max(0, scroll_y - 30)
            elif event.button == 5:  # Scroll Down
                scroll_y = min(max_scroll, scroll_y + 30)
                
            # Direct Interaction checking inside target coordinates
            elif event.button == 1:
                for emp in manager.employees:
                    if emp.coffee_btn.collidepoint(event.pos) and manager.money >= emp.coffee_cost:
                        manager.pay_wages(emp.coffee_cost)
                        emp.is_resting = True

    # Process State Machines
    manager.update(dt)

    # Frame Buffer Clean
    screen.fill(COLOR_BG)

    # Infinite Grid Draw Compiler Logic Loops
    col_width, row_height = 270, 210
    start_x, start_y = 50, 160
    
    for i, employee in enumerate(manager.employees):
        column = i % 3
        row = i // 3
        pos_x = start_x + (column * col_width)
        pos_y = start_y + (row * row_height)
        
        # Render item instances using our calculated layout and scroll metrics
        employee.draw(screen, pos_x, pos_y, scroll_y, manager.current_hour)
        
    # Recalculate deep scroll limits dynamically as data changes
    total_rows = ((len(manager.employees) - 1) // 3) + 1
    max_scroll = max(0, (start_y + (total_rows * row_height)) - (HEIGHT - 50))

    # Sticky Persistent HUD Layer Dashboard Overlay
    pygame.draw.rect(screen, COLOR_SURFACE, (0, 0, WIDTH, 130))
